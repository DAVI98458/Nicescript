local function getHRP()
    return LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
end

local function getPartCFrame(prompt)
    local parent = prompt.Parent
    if parent and parent:IsA("BasePart") then return parent.CFrame end
    if parent and parent:IsA("Model") and parent.PrimaryPart then return parent.PrimaryPart.CFrame end
    if parent then
        local gp = parent.Parent
        if gp and gp:IsA("BasePart") then return gp.CFrame end
        if gp and gp:IsA("Model") and gp.PrimaryPart then return gp.PrimaryPart.CFrame end
    end
    return nil
end

local function TeleportTo(prompt)
    local hrp = getHRP()
    if not hrp then return end
    local cf = getPartCFrame(prompt)
    if not cf then return end
    hrp.CFrame = cf + Vector3.new(0, 3, 0)
end

local function TeleportToExact(prompt)
    local hrp = getHRP()
    if not hrp then return end
    -- Teleporta direto na posição mundial do ProximityPrompt com offset pra cima
    local part = prompt.Parent
    if part and part:IsA("BasePart") then
        hrp.CFrame = part.CFrame * CFrame.new(0, 2, 0)
    else
        local cf = getPartCFrame(prompt)
        if not cf then return end
        hrp.CFrame = cf * CFrame.new(0, 2, 0)
    end
end

local function TeleportToPos(x, y, z)
    local hrp = getHRP()
    if not hrp then return end
    hrp.CFrame = CFrame.new(x, y, z)
end

local function FirePrompt(prompt)
    if fireproximityprompt then fireproximityprompt(prompt) end
end

local function getOrigin()
    local hrp = getHRP()
    return hrp and hrp.CFrame
end

local function returnTo(origin)
    local hrp = getHRP()
    if hrp and origin then hrp.CFrame = origin end
end

local function findItem(name)
    if LP.Character then
        for _, item in ipairs(LP.Character:GetChildren()) do
            if item.Name == name then return item, true end
        end
    end
    if LP:FindFirstChild("Backpack") then
        for _, item in ipairs(LP.Backpack:GetChildren()) do
            if item.Name == name then return item, false end
        end
    end
    return nil, false
end

local function equipItem(name)
    local item, jaEquipado = findItem(name)
    if not item then
        WindUI:Notify({ Title = "Erro", Content = "'"..name.."' not in the inventory!", Icon = "x-circle", Duration = 3 })
        return false
    end
    if not jaEquipado then
        local humanoid = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
        if humanoid then humanoid:EquipTool(item) task.wait(0.15) end
    end
    return true
end

local function safeWFC(parent, name, timeout)
    timeout = timeout or 5
    return parent:WaitForChild(name, timeout)
end

-- Função reutilizável de collect (usada em Geral e Task)
local function doCollectTrashes()
    local origin = getOrigin()
    local game_ = safeWFC(workspace, "Game")
    if not game_ then warn("[CollectTrash] Game não encontrado!") return end
    local trashFolder = safeWFC(game_, "trashes")
    if not trashFolder then warn("[CollectTrash] trashes não encontrado!") return end
    local done = {}
    for _, obj in ipairs(trashFolder:GetDescendants()) do
        if obj:IsA("ProximityPrompt") and not done[obj] then
            done[obj] = true
            TeleportTo(obj)
            task.wait(0.2)
            FirePrompt(obj)
            task.wait(0.2)
        end
    end
    returnTo(origin)
end

-- AUTO CLOSE CURTAINS
local autoCurtainsEnabled = false
local curtainConn = nil
local curtainRunning = false

local CURTAIN_TARGETS = {
    {index = 2,   name = nil},
    {index = 3,   name = nil},
    {index = 4,   name = nil},
    {index = 5,   name = nil},
    {index = 6,   name = nil},
    {index = 7,   name = nil},
    {index = nil, name = "CurtainN"},
}

local function findPromptIn(parent)
    for j = 1, 50 do
        local child = parent:FindFirstChild(tostring(j))
        if child and child:IsA("ProximityPrompt") then return child end
    end
    for _, obj in ipairs(parent:GetDescendants()) do
        if obj:IsA("ProximityPrompt") then return obj end
    end
    return nil
end

local function runCurtainLoop()
    if curtainRunning then return end
    curtainRunning = true

    local origin = getOrigin()
    local house = workspace:FindFirstChild("House")
    if not house then curtainRunning = false return end
    local curtains = house:FindFirstChild("Curtains")
    if not curtains then curtainRunning = false return end
    local children = curtains:GetChildren()

    local keepGoing = true
    while keepGoing and autoCurtainsEnabled do
        keepGoing = false

        for _, target in ipairs(CURTAIN_TARGETS) do
            if not autoCurtainsEnabled then break end

            local curtain = target.name and curtains:FindFirstChild(target.name)
                or (target.index and children[target.index])
            if not curtain then continue end

            local prompt = curtain:FindFirstChild("Prompt")
            if not prompt then continue end
            local open = prompt:FindFirstChild("Open")
            if not open then continue end

            if open.Value ~= true then continue end

            keepGoing = true

            local pp = findPromptIn(prompt)
            if pp then
                TeleportTo(pp)
                task.wait(0.15)
                FirePrompt(pp)
                task.wait(0.15)
            end

            break
        end
    end

    returnTo(origin)
    curtainRunning = false
end

local function connectCurtains()
    local house = safeWFC(workspace, "House")
    if not house then return end
    local curtains = safeWFC(house, "Curtains")
    if not curtains then return end
    local children = curtains:GetChildren()

    for _, target in ipairs(CURTAIN_TARGETS) do
        local curtain = target.name and curtains:FindFirstChild(target.name)
            or (target.index and children[target.index])
        if not curtain then continue end
        local prompt = curtain:FindFirstChild("Prompt")
        if not prompt then continue end
        local open = prompt:FindFirstChild("Open")
        if not open then continue end

        open:GetPropertyChangedSignal("Value"):Connect(function()
            if open.Value == true and autoCurtainsEnabled then
                task.spawn(runCurtainLoop)
            end
        end)
    end

    task.spawn(runCurtainLoop)
end

TabAuto:Toggle({
    Title = "Auto Close Curtains",
    Description = "It closes the curtains automatically when they are opened",
    Icon = "eye-off",
    Value = false,
    Callback = function(state)
        autoCurtainsEnabled = state
        if state then
            task.spawn(connectCurtains)
        else
            if curtainConn then curtainConn:Disconnect() curtainConn = nil end
        end
    end,
})

-- AUTO SLEEP
local autoSleepEnabled = false
local autoSleepConn = nil
local autoSleepDone = false

local function connectAutoSleep()
    local chaseDoor = workspace:WaitForChild("Game", 10)
        :WaitForChild("dad", 10)
        :WaitForChild("PossesedDad", 10)
        :WaitForChild("ChaseDoor", 10)
    if not chaseDoor then warn("[AutoSleep] ChaseDoor não encontrado!") return end

    local BED_TARGETS = {
        {index = 2,   name = nil},
        {index = 3,   name = nil},
        {index = 4,   name = nil},
        {index = 5,   name = nil},
        {index = nil, name = "Bed"},
    }

    local function doSleep()
        local bedsFolder = workspace.House.Rooms.Bedroom:FindFirstChild("Beds")
        if not bedsFolder then warn("[AutoSleep] Beds não encontrado!") return end

        local children = bedsFolder:GetChildren()

        for _, target in ipairs(BED_TARGETS) do
            local bed = target.name and bedsFolder:FindFirstChild(target.name)
                or (target.index and children[target.index])
            if not bed then continue end

            local primary = bed:FindFirstChild("Primary")
            if not primary then continue end
            local pp = primary:FindFirstChild("ProximityPrompt")
            if not pp then continue end
            if not pp.Enabled then continue end

            TeleportToExact(pp)
            task.wait(0.2)
            FirePrompt(pp)
            return
        end

        warn("[AutoSleep] Nenhum ProximityPrompt ativo em nenhuma cama!")
    end

    if chaseDoor.Value == true then
        autoSleepDone = true
        doSleep()
    end

    autoSleepConn = chaseDoor:GetPropertyChangedSignal("Value"):Connect(function()
        if not autoSleepEnabled then return end
        if chaseDoor.Value == true then
            if autoSleepDone then return end
            autoSleepDone = true
            doSleep()
        else
            autoSleepDone = false
        end
    end)
end

TabAuto:Toggle({
    Title = "Auto Sleep",
    Description = "When Dad wakes up, he'll automatically go to bed",
    Icon = "moon",
    Value = false,
    Callback = function(state)
        autoSleepEnabled = state
        if state then
            autoSleepDone = false
            task.spawn(connectAutoSleep)
        else
            if autoSleepConn then autoSleepConn:Disconnect() autoSleepConn = nil end
            autoSleepDone = false
        end
    end,
})

        hum.WalkSpeed = state and selectedSpeed or 16
    end,
})

local instantPromptEnabled = false
local promptConnections = {}

local function applyInstantPrompts()
    for _, conn in ipairs(promptConnections) do conn:Disconnect() end
    promptConnections = {}
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("ProximityPrompt") then obj.HoldDuration = 0 end
    end
    local conn = workspace.DescendantAdded:Connect(function(obj)
        if instantPromptEnabled and obj:IsA("ProximityPrompt") then obj.HoldDuration = 0 end
    end)
    table.insert(promptConnections, conn)
end

local function removeInstantPrompts()
    for _, conn in ipairs(promptConnections) do conn:Disconnect() end
    promptConnections = {}
end

TabPlayer:Toggle({
    Title = "Activate ProximityPrompt Instant",
    Description = "Very Fast interact",
    Icon = "mouse-pointer-click",
    Value = false,
    Callback = function(state)
        instantPromptEnabled = state
        if state then applyInstantPrompts() else removeInstantPrompts() end
    end,
})

        hum.WalkSpeed = selectedSpeed
    end
